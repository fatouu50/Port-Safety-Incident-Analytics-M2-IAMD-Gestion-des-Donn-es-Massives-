# Projet 11 : chargement de l'archive dans HDFS (à lancer depuis 05_HDFS)
# Prérequis : docker compose up -d   puis   python prepare_archive.py

# 1. PC -> système local du conteneur namenode
docker cp archive namenode:/tmp/archive

# 2. Créer le dossier du projet dans HDFS
docker exec namenode hdfs dfs -mkdir -p /port_exam/project_11

# 3. Conteneur -> HDFS (-f : remplace si déjà présent)
docker exec namenode hdfs dfs -put -f /tmp/archive /port_exam/project_11/

# 4. Vérifier
docker exec namenode hdfs dfs -ls -R /port_exam/project_11
docker exec namenode hdfs dfs -du -h /port_exam/project_11/archive
docker exec namenode hdfs dfs -count /port_exam/project_11/archive

# 5. Blocs et réplication
docker exec namenode hdfs fsck /port_exam/project_11/archive
docker exec namenode hdfs fsck /port_exam/project_11/archive/raw/incidents/year=2026/month=05/incidents_2026_05.csv -files -blocks -locations
docker exec namenode hdfs dfs -stat "%r %o %b %n" /port_exam/project_11/archive/raw/incidents/year=2026/month=05/incidents_2026_05.csv
docker exec namenode hdfs dfs -setrep -w 2 /port_exam/project_11/archive

# 6. Tolérance aux pannes : un DataNode arrêté, le fichier reste lisible
docker stop datanode1
docker exec namenode hdfs dfs -cat /port_exam/project_11/archive/reference/locations.csv
docker start datanode1

# 7. Vérification HDFS = manifeste (lignes et MD5)
python hdfs_verify.py

# 8. Tables MART en Parquet (après 04_ETL/etl_pipeline.py)
python export_mart_parquet.py
docker exec namenode rm -rf /tmp/curated
docker cp archive/curated namenode:/tmp/curated
docker exec namenode hdfs dfs -put -f /tmp/curated /port_exam/project_11/archive/
docker exec namenode hdfs dfs -ls /port_exam/project_11/archive/curated/mart
python hdfs_verify.py
